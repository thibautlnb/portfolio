Script 1 — md_to_html.py
md_to_html.py Convertisseur Markdown → HTML via la bibliothèque markdown ▼
import markdown open() / read() encodage UTF-8
Fonction utilitaire réutilisée par les deux autres scripts. Lit un fichier .md, le convertit en HTML via markdown.markdown() et écrit le résultat dans un fichier .html portant le même nom. Constitue la brique de base du pipeline. 
import markdown

def convert(nom):
    "Convertit un fichier Markdown en HTML"

    with open(nom + ".md", 'r', encoding="utf-8") as f:
        text = f.read()

    html = markdown.markdown(text)   # conversion MD → HTML

    with open(nom + ".html", 'w', encoding="utf-8") as f:
        f.write(html)
Script 2 — fiche_osm.py
fiche_osm.py Génère une fiche HTML d'un nœud OpenStreetMap à partir de son identifiant ▼
requests.get() OSM API v0.6 XML → JSON dict iteration
Interroge l'API REST d'OpenStreetMap pour récupérer les métadonnées d'un nœud (id, lat, lon, tags). Construit un fichier Markdown structuré avec tous les attributs, puis appelle convert() pour produire le HTML final. 
import requests
from md_to_html import convert

def get_node(id: int) -> dict:
    """ Récupère les infos JSON d'un nœud OSM via son id """
    data = requests.get(
        f"https://www.openstreetmap.org/api/0.6/node/{id}.json"
    )
    return data.json()

def node_to_md(data: dict, filename: str) -> None:
    """ Écrit les données JSON en Markdown """
    data2 = data["elements"][0]
    data3 = data2["tags"]

    with open(filename, 'w') as fichier:
        fichier.write("# " + data3["name"] + "\n")
        for cle, valeur in data2.items():   # id, lat, lon…
            if cle != "tags":
                fichier.write(f"* {cle} : {valeur}\n")
        for cle, valeur in data3.items():   # tags OSM
            if cle != "name":
                fichier.write(f"* {cle} : {valeur}\n")

def fiche_osm(id: int) -> None:
    node_to_md(get_node(id), "fichier.md")
    convert("fichier")   # → fichier.html
Script 3 — infos_locales.py (script principal)
infos_locales.py Analyse complète des arrêts de bus d'une ville — statistiques + rapport Markdown/HTML ▼
Overpass API requests.post() dict / list f-strings sys.argv boucles for conditions if
Script principal du projet. Envoie une requête Overpass QL pour récupérer tous les arrêts de bus (highway=bus_stop) d'une ville via son article Wikipédia. Calcule 7 statistiques (abris, bancs, poubelles, éclairages, accessibilité handicapé, dalles podotactiles, horaires affichés) avec des comptages et pourcentages. Génère ensuite un rapport Markdown détaillé arrêt par arrêt, puis le convertit en HTML. 
def get_dataset(ville: str) -> dict:
    """Requête Overpass pour tous les bus_stop de la ville"""
    requete = f"""
    [out:json][timeout:300];
    area["wikipedia"="fr:{ville}"]->.zone;
    nwr["highway"="bus_stop"](area.zone);
    out;
    """
    url = "https://overpass-api.de/api/interpreter"
    response = requests.post(url, data={'data': requete})
    return response.json()

def compute_statistics(dataset: dict) -> dict:
    """Calcule 7 indicateurs sur les arrêts de bus"""
    elements = dataset.get('elements', [])
    total = len(elements)
    nb_abris = nb_bancs = nb_poubelles = 0
    nb_eclairages = nb_handicap = nb_dalles = 0

    for obj in elements:
        tags = obj.get('tags', {})
        if tags.get("shelter") == "yes": nb_abris += 1
        if tags.get("bench")   == "yes": nb_bancs += 1
        if tags.get("bin")     == "yes": nb_poubelles += 1
        if tags.get("lit")     == "yes": nb_eclairages += 1
        if tags.get("wheelchair") == "yes": nb_handicap += 1
        if tags.get("tactile_paving") == "yes": nb_dalles += 1

    return {
        "Nombre d'arrêts"  : total,
        "Abris"            : f"{nb_abris} ({nb_abris*100//total}%)",
        "Bancs"            : f"{nb_bancs} ({nb_bancs*100//total}%)",
        "Éclairage"        : f"{nb_eclairages} ({nb_eclairages*100//total}%)",
        "Accessibilité PMR": f"{nb_handicap} ({nb_handicap*100//total}%)",
    }

def infos_locales(ville: str, filename: str) -> None:
    """Chaîne complète : API → stats → Markdown → HTML"""
    donnees = get_dataset(ville)
    dataset_to_md(donnees, filename, ville)
    convert(filename)   # appel md_to_html.py

# Utilisation en ligne de commande :
# python infos_locales.py Caen rapport_caen
